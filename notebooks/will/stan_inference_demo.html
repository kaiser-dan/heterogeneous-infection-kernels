```{python}
def cap_exposure_bins(M, N, max_k):
    """
    Collapse exposure matrices to `max_k` bins by pooling any counts above the cap
    into the final bin (>= max_k).
    """
    if max_k is None:
        return M, N
    if max_k < 1:
        raise ValueError("max_k must be >= 1")

    def _cap(arr):
        if arr is None:
            return None
        arr = np.asarray(arr)
        if arr.ndim != 2:
            raise ValueError("Exposure arrays must be 2D")
        # Already within the requested cap
        if arr.shape[1] <= max_k:
            return arr[:, :max_k]
        capped = np.zeros((arr.shape[0], max_k), dtype=arr.dtype)
        if max_k > 1:
            capped[:, :max_k - 1] = arr[:, :max_k - 1]
        capped[:, -1] = arr[:, max_k - 1:].sum(axis=1)
        return capped

    return _cap(M), _cap(N)
```

## Simulated Dynamics Sanity Check

Generate synthetic M/N from a simulated contagion on the same network and run the Stan inference to verify the pipeline.

```{python}
#| label: simulate-dynamics-and-fit
#| fig-cap: "Simulated contagion (heatmap + cumulative + network) and quick Stan fit."
from src.dynamics import heterogeneous_contagion_process, get_activations

# Use the same run_id's network
target_run_id = "oo_sim_165_duration"
adj_df = get_adjacency_matrix(session, target_run_id)
adj = adj_df.to_numpy(dtype=int)
N_nodes = adj.shape[0]

# Simulation params
gamma = 0.2
dt = 1.0
T_steps = 40

# Choose kmax as max binary degree (ignore edge weights) and cap to keep arrays reasonable
kmax_sim = max(1, int((adj > 0).sum(axis=1).max()))
kmax_cap = 50
kmax_sim = min(kmax_sim, kmax_cap)

# Build a simple increasing contagion kernel per node (c[i, nu] for nu>=1)
nu_grid = np.arange(1, kmax_sim + 1, dtype=float)
c_kernel = np.tile(np.clip(nu_grid / (kmax_sim + 1), 0.05, 0.9), (N_nodes, 1))

# Initial infected (few random nodes)
rng = np.random.default_rng(42)
x0 = np.zeros(N_nodes, dtype=int)
x0[rng.choice(N_nodes, size=min(3, N_nodes), replace=False)] = 1

# Simulate states over time (T x N)
```
