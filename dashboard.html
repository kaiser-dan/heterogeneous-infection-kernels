viewof selected_intervention = Inputs.select(
    interventions, 
    {value: "All Interventions", label: "Intervention"}
)

viewof selected_metric = Inputs.radio(
    ["k", "kappa"], 
    {value: "kappa", label: "Metric"}
)
viewof y_axis = Inputs.radio(
    ["Adoption Prob", "Num Retweeted", "Total Exposures"], 
    {value: "Adoption Prob"}
)
transpose = (obj) => {
  const keys = Object.keys(obj);
  return obj[keys[0]].map((_, i) => {
    const r = {};
    keys.forEach(k => r[k] = obj[k][i]);
    return r;
  });
}

// Filter and aggregate data
plot_data = {
  const all_data = transpose(stats_data);
  const data = all_data.filter(d => d.metric === selected_metric);
  
  let filtered;
  if (selected_intervention === 'All Interventions') {
    // Aggregate across all interventions
    const levels = [...new Set(data.map(d => d.level))].sort((a, b) => a - b);
    filtered = levels.map(level => {
      const level_data = data.filter(d => d.level === level);
      const adoptions = d3.sum(level_data, d => d.adoptions);
      const non_adoptions = d3.sum(level_data, d => d.non_adoptions);
      const total = adoptions + non_adoptions;
      return {
        level: level,
        adoptions: adoptions,
        non_adoptions: non_adoptions,
        total: total,
        prob: total > 0 ? adoptions / total : 0
      };
    });
  } else {
    filtered = data.filter(d => d.intervention === selected_intervention);
  }
  
  // Map to x/y for plotting
  return filtered.map(d => ({
    x: d.level,
    y: y_axis === "Adoption Prob" ? d.prob :
       y_axis === "Num Retweeted" ? d.adoptions : d.total
  }));
}

// Debug: show data structure
{
  const data = transpose(stats_data);
  const sample = data.slice(0, 3);
  return html`<div style="font-size: 0.8em; margin-bottom: 10px;">
    <details>
      <summary>Debug: Data loaded (${data.length} rows, ${plot_data.length} plot points)</summary>
      <pre>${JSON.stringify(sample, null, 2)}</pre>
    </details>
  </div>`;
}
{
  if (!plot_data || plot_data.length === 0) {
    return html`<div style="padding: 40px; text-align: center; color: #999;">
      No data available for selection
    </div>`;
  }
  
  const x_label = selected_metric === "k" ? "k (Total Exposures)" : "κ (Unique Sources)";
  
  return Plot.plot({
    width: 800,
    height: 400,
    marginBottom: 50,
    x: { label: x_label, grid: true },
    y: { label: y_axis, grid: true },
    marks: [
      Plot.barY(plot_data, {
        x: "x", 
        y: "y", 
        fill: "steelblue",
        title: d => `Level: ${d.x}\n${y_axis}: ${d.y.toFixed(4)}`
      }),
      Plot.ruleY([0])
    ]
  });
}
summary_stats = {
  const data = transpose(stats_data);
  const filtered = selected_intervention === 'All Interventions' 
    ? data 
    : data.filter(d => d.intervention === selected_intervention);
  
  const total_adoptions = d3.sum(filtered, d => d.adoptions);
  const total_exposures = d3.sum(filtered, d => d.total);
  const avg_prob = d3.mean(filtered, d => d.prob);
  
  return [
    {metric: "Total Adoptions", value: total_adoptions},
    {metric: "Total Exposures", value: total_exposures},
    {metric: "Average Adoption Prob", value: avg_prob.toFixed(4)}
  ];
}

Inputs.table(summary_stats, {
  columns: ["metric", "value"],
  header: {
    metric: "Metric",
    value: "Value"
  }
})
